 // Missing-gun exclusions still apply to every weapon.

   // Weapons without a verified unlock level are always eligible for recommendations.
   // This keeps them in the build pool even when their acquisition route is uncertain.
   if(!meta || !Number.isInteger(meta.unlockLevel)) return true;
   if(eventWeapons.has(x)) return includeEvents && own.includes(x);
   return true;
 });
 if(!candidates.length){

     ? `No owned ${cat} event weapon selected`
     : `No ${cat} weapon available at this level`;
 }
 const alreadyTags=new Set(chosen.flatMap(n=>tags[n]||[]));
 return [...candidates].sort((a,b)=>{
   const sc=n=>{
     const t=tags[n]||[];
     const prefRank=(prefs[cat]||[]).indexOf(n);
     const preference=prefRank<0?0:(prefs[cat].length-prefRank)*2.5;
     const role=t.reduce((z,x)=>z+(want.includes(x)?5:0),0);
     const modeFit=t.reduce((z,x)=>z+(modeWant.includes(x)?3:0),0);
     const coverage=t.reduce((z,x)=>z+(!alreadyTags.has(x)?2.5:0),0);
     const feedback=(feedback[n]||0)*1.5;
     const versatility=new Set(t).size*0.8;
     return preference+role+modeFit+coverage+feedback+versatility;
   };
   return sc(b)-sc(a);
 })[0];
}
