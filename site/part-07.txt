async function loadFullWeaponDatabase(){
  const next={Primary:[],Backup:[],Melee:[],Special:[],Sniper:[],Heavy:[]};
  const api=async(cat)=>{
    const url=`https://pg3d.wiki.gg/api.php?action=query&list=categorymembers&cmtitle=${encodeURIComponent("Category:"+wikiCategoryMap[cat])}&cmlimit=500&cmnamespace=0&format=json&origin=*`;
    const r=await fetch(url,{cache:"no-store"});
    if(!r.ok) throw new Error("wiki request failed");
    const j=await r.json();
    return (j.query?.categorymembers||[]).map(x=>x.title).filter(Boolean);
  };
  try{
    const results=await Promise.all(Object.keys(next).map(api));
    Object.keys(next).forEach((cat,i)=>{
      next[cat]=[...new Set(results[i].filter(w=>!/^Category:|\(PGW\)| Up[123] \(PGW\)$/.test(w)))];
    });
    Object.keys(next).forEach(cat=>{db[cat]=[...new Set([...next[cat],...(db[cat]||[])])];});
    databaseLoaded=true;
    for(const name of Object.values(db).flat()) if(!weaponUnlockData[name]) weaponUnlockData[name]={unlockLevel:null,source:"Pixel Gun 3D Wiki category",acquisition:"Category verified; unlock level not yet verified"};
    const status=document.getElementById("levelStatus");
    if(status) status.textContent=`Full weapon database loaded from the Pixel Gun 3D Wiki: ${Object.values(db).reduce((n,a)=>n+a.length,0)} category entries. Normal Armory weapons are automatically owned; mark only the event/special weapons you own.`;
  }catch(e){
    const status=document.getElementById("levelStatus");
    if(status) status.textContent="Full Wiki database could not be loaded right now, so the built-in weapon list is being used.";
  }
  renderInventory();
}
function renderInventory(){
  if(!inv) return;
  const selected=new Set([...inv.querySelectorAll("input:checked")].map(x=>x.value));
  inv.innerHTML="";
  const all=[...new Set(Object.entries(db).flatMap(([cat,arr])=>arr.map(w=>JSON.stringify({cat,w}))))].map(x=>JSON.parse(x));
  all.sort((a,b)=>a.w.localeCompare(b.w));
  if(!all.length){
    inv.innerHTML='<div class="sub">Weapon database is empty.</div>';
    return;
  }
  const frag=document.createDocumentFragment();
  all.forEach(({cat,w})=>{
    const isEvent=eventWeapons.has(w);
    const l=document.createElement("label");
    l.className="check";
    const input=document.createElement("input");
    input.type="checkbox"; input.value=w;
    input.checked=isEvent ? selected.has(w) : true;
    input.dataset.category=cat;
    const span=document.createElement("span");
    const meta=weaponUnlockData[w]||{};
    const levelText=Number.isInteger(meta.unlockLevel)?`Level ${meta.unlockLevel}+`:"Level unverified";
    span.textContent=`${cat} — ${w} • ${isEvent?"Special/Event — mark if owned":"Armory — automatically owned"} • ${levelText}`;
    l.append(input,span); frag.appendChild(l);
  });
  inv.appendChild(frag);
  const status=document.getElementById("levelStatus");
  if(status) status.textContent=`All ${all.length} weapon entries are listed. Armory weapons are pre-owned automatically; special/event weapons can be checked manually. Unknown-level weapons remain eligible for recommendations.`;
}
