function geminiKey(){
  const input=document.getElementById("geminiApiKey");
  const typed=(input?.value||"").trim();
  if(typed){sessionStorage.setItem("geminiApiKey",typed);return typed;}
  return sessionStorage.getItem("geminiApiKey")||"";
}
function initGeminiKeyUI(){
  const input=document.getElementById("geminiApiKey");
  const stored=sessionStorage.getItem("geminiApiKey")||"";
  if(input&&stored) input.value=stored;
  document.getElementById("clearGeminiKey")?.addEventListener("click",()=>{

    if(input) input.value="";
    const s=document.getElementById("geminiStatus");
    if(s)s.textContent="Gemini key cleared from this browser session.";

}
function geminiWeaponProfile(category,name){
  const stats=window.weaponStats?.[name]||{};
  const attrs=String(stats.attrs||"").replace(/\\s+/g," ").trim().slice(0,150);
  const text=(name+" "+attrs).toLowerCase();
  const roles=[];
  const categoryRoles={
    Primary:"main sustained-damage weapon",
    Backup:"fast-switch backup and close-fight coverage",
    Melee:"close-range damage and movement utility",
    Special:"utility, control, defense, or support",
    Sniper:"long-range precision and burst damage",
    Heavy:"area damage, explosive pressure, or heavy burst"
  };
  if(/automatic|minigun|stream beam|rapid fire/.test(text))roles.push("sustained or rapid fire");
  if(/shotgun|flamethrower|melee|chainsaw/.test(text))roles.push("close range");
  if(/sniper|crossbow|railgun|marksman|scope|long range/.test(text))roles.push("long range");
  if(/rocket|grenade|bomb|mortar|launcher|area damage|splash|bazooka|missile/.test(text))roles.push("area or splash pressure");
  if(/poison|burn|freeze|slow|stun|blind|pull|gravity|disabl/.test(text))roles.push("status or crowd control");
  if(/heal|healing|shield|barrier|support|buff|ammo restore/.test(text))roles.push("support or defense");
  if(/dash|teleport|mobility|double jump|jet/.test(text))roles.push("mobility");
  if(/silent|stealth|ghost/.test(text))roles.push("stealth");
  if(name==="Crystal Laser Cannon")roles.push("primary automatic; fast sustained fire; medium range; high mobility; magazine empties quickly");
  const numeric={};
  ["damage","fireRate","capacity","mobility"].forEach(key=>{
    const value=Number(stats[key]);
    if(Number.isFinite(value)&&value>0)numeric[key]=value;
  });
  return {category,role:categoryRoles[category]||category,traits:roles,stats:numeric,attributes:attrs||undefined};
}
function geminiResponseText(data){
  const steps=Array.isArray(data?.steps)?data.steps:[];
  return steps
    .filter(step=>step?.type==="model_output")
    .flatMap(step=>Array.isArray(step.content)?step.content:[])
    .filter(part=>part?.type==="text"&&typeof part.text==="string")
    .map(part=>part.text)
    .join("\\n").trim();
}
