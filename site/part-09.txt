<script id="missingGunTrackerFix">
(function(){
 const KEY="loadoutMissingGuns";
 function read(){
   let a=[],b=[];
   try{a=JSON.parse(localStorage.getItem(KEY)||"[]")}catch(e){}
   try{b=JSON.parse(localStorage.getItem("loadoutMissing")||"[]")}catch(e){}
   return [...new Set([...a,...b].filter(x=>typeof x==="string"))];
 }
 function save(list){
   const clean=[...new Set(list)];
   localStorage.setItem(KEY,JSON.stringify(clean));
   localStorage.setItem("loadoutMissing",JSON.stringify(clean));
 }
 window.renderMissingGunList=function(){
   let box=document.getElementById("missingGunList");
   if(!box){
     box=document.createElement("section");
     box.id="missingGunList";box.className="missing-gun-list";
     const result=document.getElementById("output")||document.querySelector(".results")||document.body;
     result.appendChild(box);
   }
   const missing=read();
   box.innerHTML="";
   const heading=document.createElement("h3");
   heading.textContent="Guns You Said You Don't Have ("+missing.length+")";
   box.appendChild(heading);
   if(!missing.length){
     const empty=document.createElement("p");
     empty.textContent="No guns marked missing yet. Click “I don't have this gun” on a recommendation.";
     box.appendChild(empty);return;
   }
   const ul=document.createElement("ul");
   missing.forEach(name=>{
     const li=document.createElement("li");
     const label=document.createElement("span");label.textContent=name;
     const btn=document.createElement("button");btn.type="button";btn.textContent="I have this gun now";
     btn.addEventListener("click",()=>window.restoreMissingGun(name));
     li.append(label,btn);ul.appendChild(li);
   });
   box.appendChild(ul);
 };
 window.markMissingGun=function(name){
   if(!name)return;
   save([...read(),name]);
   document.querySelectorAll(".slot").forEach(card=>{
     if(card.querySelector("h3")?.textContent===name){
       const btn=card.querySelector(".missing-btn");
       if(btn)btn.textContent="Added to missing-gun list";
     }
   });
   window.renderMissingGunList();
   const status=document.getElementById("status");
   if(status)status.textContent="MISSING GUN SAVED";
 };
 window.restoreMissingGun=function(name){
   save(read().filter(n=>n!==name));
   window.renderMissingGunList();
   const status=document.getElementById("status");
   if(status)status.textContent="GUN RESTORED — REGENERATE LOADOUT";
 };
 window.clearMissingGuns=function(){
   save([]);
   window.renderMissingGunList();
   const status=document.getElementById("status");
   if(status)status.textContent="MISSING GUN LIST CLEARED";
 };
 document.addEventListener("DOMContentLoaded",window.renderMissingGunList);
 window.addEventListener("load",window.renderMissingGunList);
})();
</script>
